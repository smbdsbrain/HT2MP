if(TARGET enet::enet)
    set(HT2MP_ENET_TARGET enet::enet)
elseif(TARGET enet)
    set(HT2MP_ENET_TARGET enet)
else()
    message(FATAL_ERROR "ht2mp_net requires an existing ENet target named enet or enet::enet")
endif()

add_library(ht2mp_net STATIC client.cpp)
target_include_directories(ht2mp_net PUBLIC "${PROJECT_SOURCE_DIR}/include")
if(DEFINED enet_SOURCE_DIR)
    target_include_directories(ht2mp_net SYSTEM PRIVATE "${enet_SOURCE_DIR}/include")
else()
    find_path(HT2MP_ENET_INCLUDE_DIR enet/enet.h REQUIRED)
    target_include_directories(ht2mp_net SYSTEM PRIVATE "${HT2MP_ENET_INCLUDE_DIR}")
endif()
target_link_libraries(ht2mp_net PUBLIC ht2mp_protocol PRIVATE ${HT2MP_ENET_TARGET})
target_compile_features(ht2mp_net PUBLIC cxx_std_20)

if(MSVC)
    target_compile_options(ht2mp_net PRIVATE /W4 /permissive- /EHsc /FS)
    target_compile_definitions(ht2mp_net PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    set_property(TARGET ht2mp_net PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
