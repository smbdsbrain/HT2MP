if(TARGET enet::enet)
    set(HT2MP_ENET_TARGET enet::enet)
elseif(TARGET enet)
    set(HT2MP_ENET_TARGET enet)
else()
    message(FATAL_ERROR "ht2mp-sim requires an existing ENet target named enet or enet::enet")
endif()

add_library(ht2mp_sim_protocol STATIC server_protocol.cpp)
target_include_directories(ht2mp_sim_protocol PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(ht2mp_sim_protocol PUBLIC ht2mp_protocol)
target_compile_features(ht2mp_sim_protocol PUBLIC cxx_std_20)

add_executable(ht2mp-sim main.cpp)
if(DEFINED enet_SOURCE_DIR)
    target_include_directories(ht2mp-sim SYSTEM PRIVATE "${enet_SOURCE_DIR}/include")
else()
    find_path(HT2MP_ENET_INCLUDE_DIR enet/enet.h REQUIRED)
    target_include_directories(ht2mp-sim SYSTEM PRIVATE "${HT2MP_ENET_INCLUDE_DIR}")
endif()
target_link_libraries(ht2mp-sim PRIVATE ht2mp_sim_protocol ${HT2MP_ENET_TARGET})
target_compile_features(ht2mp-sim PRIVATE cxx_std_20)

if(MSVC)
    target_compile_options(ht2mp_sim_protocol PRIVATE /W4 /permissive- /EHsc /FS)
    target_compile_definitions(ht2mp_sim_protocol PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    set_property(TARGET ht2mp_sim_protocol PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(ht2mp-sim PRIVATE /W4 /permissive- /EHsc /FS)
    target_compile_definitions(ht2mp-sim PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    set_property(TARGET ht2mp-sim PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
